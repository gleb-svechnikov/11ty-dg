% Obsidian callouts: shaded box with a coloured label
\newsavebox{\calloutbox}
\newenvironment{callout}[2]{%
  \def\calloutcolor{#1}%
  \par\medskip\noindent
  \begin{lrbox}{\calloutbox}%
  \begin{minipage}{\dimexpr\linewidth-2\fboxsep\relax}\small
  \setlength{\parskip}{3pt}%
  {\bfseries\color{#1}#2.}\par
}{%
  \end{minipage}\end{lrbox}%
  \colorbox{\calloutcolor!8}{\usebox{\calloutbox}}\par\medskip
}
